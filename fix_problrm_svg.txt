normalizeAyahData.ts

export interface RawAyahEntry {
surah?: number;
sura_number?: number;
ayah?: number;
ayah_number?: number;
polygon?: string;
path?: string;
d?: string;
[key: string]: any;
}

export interface NormalizedAyah {
surah: number;
ayah: number;
geometry: {
type: 'path';
d: string;
};
}

export interface NormalizedPageSchema {
page: number;
viewBox: string; // أو مصفوفة أبعاد
ayahs: NormalizedAyah[];
}

/**
• دالة التحويل الموحد (Normalizer)
• تحول أي صيغة إحداثيات (Polygon Points أو SVG Path) إلى Schema موحد تماماً
*/
export function normalizeAyahData(rawPageData: any, pageNumber: number): NormalizedPageSchema {
// 1. استخراج مصفوفة الآيات من أي هيكل محتمل في ملف الـ JSON
const rawAyahs: RawAyahEntry[] =
rawPageData.ayahs ||
rawPageData.lines ||
rawPageData.data ||
(Array.isArray(rawPageData) ? rawPageData : []);

const normalizedAyahs: NormalizedAyah[] = [];

for (const item of rawAyahs) {
const surah = Number(item.surah || item.sura_number || 0);
const ayah = Number(item.ayah || item.ayah_number || 0);

let svgPathD = '';

// الحالة الأولى: الصيغة الأولى (Points مثل "181.08,18.31 57.54,18.31...")
if (item.polygon && !item.polygon.trim().startsWith('M')) {
svgPathD = convertPointsToSvgPath(item.polygon);
}
// الحالة الثانية: صيغة SVG Path المباشرة (مثل "M 0.0 0.75 L 337.75 ... Z")
else if (item.path || item.d || item.polygon) {
svgPathD = item.path || item.d || item.polygon;
}

if (surah > 0 && ayah > 0 && svgPathD) {
normalizedAyahs.push({
surah,
ayah,
geometry: {
type: 'path',
d: svgPathD,
},
});
}
}

return {
page: pageNumber,
viewBox: rawPageData.viewBox || "0 0 1000 1450", // الأبعاد الافتراضية لمصحف kfqc
ayahs: normalizedAyahs,
};
}

/**
• دالة مسؤولة حصرياً عن تحويل إحداثيات النقاط (Points) إلى SVG Path (M ... L ... Z)
• لا تقترب من مسارات Path الأصلية نهائياً لتجنب تكسير أوامر C و Q و A
*/
function convertPointsToSvgPath(pointsString: string): string {
const points = pointsString
.trim()
.split(/\s+/)
.map((p) => p.split(',').map(Number));

if (points.length === 0 || points[0].length < 2) return '';

let pathD = M ${points[0][0]} ${points[0][1]};
for (let i = 1; i < points.length; i++) {
pathD += L ${points[i][0]} ${points[i][1]};
}
pathD += ' Z'; // إغلاق المضلع

return pathD;
}

Important_problem .txt
لكن لا ينبغي أن نفترض أن الزخرفة هي السبب دون فحص قيمة viewBox الفعلية داخل SVG.
مثلاً:
XML
<svg viewBox="-53.31 -198.47 400 600">
هذا يعني أن نقطة بداية نظام الإحداثيات هي:
minX = -53.31
minY = -198.47
وهذه القيم يجب أن تبقى محفوظة.
لا نحولها إلى صفر، ولا نعدل إحداثيات JSON بشكل عشوائي.
تصحيح معادلة Touch Mapping
المعادلة الموجودة في ملفك:
SVG X = Touch X × ViewBox Width / Component Width
ليست كافية في جميع الحالات.
لماذا؟
لأننا نحتاج إلى مراعاة:

المعادلة الصحيحة لهذا الإعداد:
لنفترض:
W: عرض مكون SVG على الشاشة.
H: ارتفاع المكون.

vbH: ارتفاع ViewBox.
نحسب:
const scale = Math.min(
W / vbW,
H / vbH
);

const offsetX = (W - vbW * scale) / 2;
const offsetY = (H - vbH * scale) / 2;

const svgX =
vbX + (locationX - offsetX) / scale;

const svgY =
vbY + (locationY - offsetY) / scale;
استخدم locationX وlocationY من حدث اللمس، بشرط أن تكون إحداثيات الحدث نسبية إلى مكون SVG نفسه.
